/* ====================================================================== CODE.GS — Confirmação automática de carga (Câmara Fria · Bread King) O QUE ISSO FAZ: Recebe uma lista de paletes (endereço completo: rua, lado, andar, posição, palete) enviada pelo painel web, encontra essas linhas na aba "Ocupação" e preenche a coluna DATA SAÍDA com a data de hoje — só nas linhas que ainda não tinham data de saída (pra não sobrescrever histórico por engano). COMO INSTALAR: siga o passo a passo que o Claude te mandou junto com este arquivo. Resumo rápido: 1) Abra a planilha "Ocupação" no Google Sheets. 2) Extensões → Apps Script. 3) Apague o conteúdo padrão e cole ESTE arquivo inteiro. 4) Troque o valor de TOKEN_ESPERADO abaixo por uma senha sua. 5) Implantar → Nova implantação → tipo "App da Web". - Executar como: Eu (sua conta) - Quem pode acessar: Qualquer pessoa 6) Copie a URL gerada (termina em /exec) e cole em APPS_SCRIPT_URL no arquivo do painel (index.html). 7) Cole essa MESMA senha em APPS_SCRIPT_TOKEN no index.html. ====================================================================== */ // TROQUE por uma senha sua — precisa ser IGUAL à constante // APPS_SCRIPT_TOKEN configurada no painel (index.html) const TOKEN_ESPERADO = 'bk-camara-2026-x7'; // Nome exato da aba de ocupação na planilha const NOME_ABA_OCUPACAO = 'CAMARA 1'; // Nome exato da aba de referência de famílias (FAMILIA, CÓDIGO, PRODUTO, RUA) const NOME_ABA_FAMILIAS = 'FAMILIAS'; // ID da planilha (o trecho entre /d/ e /edit na URL dela no navegador). // Isso é necessário porque, dentro de um Web App (doPost), não existe // "planilha ativa" — getActiveSpreadsheet() só funciona quando alguém // está com a planilha aberta na tela, o que não é o caso aqui. const SPREADSHEET_ID = '1e2pq-mag8NsODFM_2SUXfOoYF5_zLbdIbw0Ipf6w4wQ'; function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.token !== TOKEN_ESPERADO) { return respostaJson({ ok: false, erro: 'Senha (token) inválida.' }); } const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const aba = ss.getSheetByName(NOME_ABA_OCUPACAO); if (!aba) { const nomesExistentes = ss.getSheets().map(function(s){ return s.getName(); }).join(', '); return respostaJson({ ok: false, erro: 'Aba "' + NOME_ABA_OCUPACAO + '" não encontrada. Abas existentes: ' + nomesExistentes }); } const acao = body.acao || 'retirada'; // por padrão, sem "acao" = confirmação de carga (retirada) if (acao === 'entrada') return processarEntrada(aba, body); return processarRetirada(aba, body); } catch (err) { return respostaJson({ ok: false, erro: String(err) }); } } /* ação "retirada" (aba Montar Carga): pra cada palete confirmado — "completo" preenche a DATA SAÍDA de hoje (a posição libera); "parcial" (só acontece em palete misto antigo, formato de vários produtos numa linha só) remove os produtos indicados da coluna PRODUTO daquela linha, sem mexer na DATA SAÍDA; "pacotes_parcial" reduz o número da coluna PACOTES daquela linha pela quantidade pedida — a posição continua ocupada e o palete não sai do lugar, até os pacotes chegarem a zero, quando então vira uma retirada completa (preenche DATA SAÍDA). */ /* Grava a DATA SAÍDA forçando a célula pro formato texto antes — sem isso, o Google Sheets pode "entender" a string como data/hora e reformatar a exibição conforme o formato numérico da coluna, podendo esconder a hora (ou trocar a ordem dia/mês) na exportação em CSV que o painel lê. */ function gravarDataSaida(aba, linha, idx, valor){ const cel = aba.getRange(linha + 1, idx.dataSaida + 1); cel.setNumberFormat('@'); cel.setValue(valor); } function processarRetirada(aba, body) { if (!Array.isArray(body.paletes) || body.paletes.length === 0) { return respostaJson({ ok: false, erro: 'Nenhum palete recebido.' }); } const dados = aba.getDataRange().getValues(); const cabecalho = dados[0]; const idx = idxColunas(cabecalho); const faltando = ['rua', 'lado', 'andar', 'posicao', 'palete', 'produto', 'dataSaida'].filter(function (k) { return idx[k] === -1; }); if (faltando.length) { return respostaJson({ ok: false, erro: 'Colunas não encontradas na planilha: ' + faltando.join(', ') }); } const precisaPacotes = body.paletes.some(function (p) { return p.tipo === 'pacotes_parcial'; }); if (precisaPacotes && idx.pacotes === -1) { return respostaJson({ ok: false, erro: 'Coluna "PACOTES" não encontrada na planilha — necessária pra retirada parcial por pacotes.' }); } const hoje = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm'); const resultados = []; body.paletes.forEach(function (p) { const linha = acharLinhaDoRegistro(dados, idx, p); const endereco = enderecoTexto(p); const produtoLabel = (p.produtosRetirar && p.produtosRetirar.length) ? p.produtosRetirar.join(', ') : (p.produto || ''); if (linha === -1) { resultados.push({ produto: produtoLabel, endereco: endereco, ok: false }); return; } if (p.tipo === 'parcial') { const produtosAtuais = String(dados[linha][idx.produto] || '').split(',').map(function (s) { return s.trim(); }).filter(Boolean); const remover = (p.produtosRetirar || []).map(function (s) { return s.trim(); }); const restantes = produtosAtuais.filter(function (prod) { return remover.indexOf(prod) === -1; }); if (restantes.length === 0) { // não sobrou produto nenhum nesse palete -> na prática é uma retirada completa gravarDataSaida(aba, linha, idx, hoje); dados[linha][idx.dataSaida] = hoje; } else { const novoValor = restantes.join(', '); aba.getRange(linha + 1, idx.produto + 1).setValue(novoValor); dados[linha][idx.produto] = novoValor; // DATA SAÍDA fica em branco de propósito: o palete continua na câmara } } else if (p.tipo === 'pacotes_parcial') { const pacotesAtuais = parseInt(dados[linha][idx.pacotes], 10); const pacotesRetirar = parseInt(p.pacotesRetirar, 10); if (!pacotesAtuais || !pacotesRetirar || pacotesRetirar <= 0) { resultados.push({ produto: produtoLabel, endereco: endereco, ok: false }); return; } const restam = pacotesAtuais - pacotesRetirar; if (restam <= 0) { // esgotou os pacotes desse produto nesse palete -> libera a posição aba.getRange(linha + 1, idx.pacotes + 1).setValue(0); gravarDataSaida(aba, linha, idx, hoje); dados[linha][idx.pacotes] = 0; dados[linha][idx.dataSaida] = hoje; } else { // ainda sobrou — só reduz o número de pacotes, palete continua na mesma posição aba.getRange(linha + 1, idx.pacotes + 1).setValue(restam); dados[linha][idx.pacotes] = restam; } } else { gravarDataSaida(aba, linha, idx, hoje); dados[linha][idx.dataSaida] = hoje; } resultados.push({ produto: produtoLabel, endereco: endereco, ok: true }); }); return respostaJson({ ok: true, resultados: resultados }); } /* ação "entrada" (aba Nova Entrada): cria linhas novas na planilha pros paletes que acabaram de chegar, já com a posição escolhida pelo painel. */ function processarEntrada(aba, body) { if (!Array.isArray(body.entradas) || body.entradas.length === 0) { return respostaJson({ ok: false, erro: 'Nenhuma entrada recebida.' }); } const cabecalho = aba.getDataRange().getValues()[0]; const idx = idxColunas(cabecalho); const faltando = ['familia', 'produto', 'rua', 'lado', 'andar', 'posicao', 'palete', 'dataProducao', 'dataEntrada', 'pacotes', 'codigoPalete'] .filter(function (k) { return idx[k] === -1; }); if (faltando.length) { return respostaJson({ ok: false, erro: 'Colunas não encontradas na planilha: ' + faltando.join(', ') }); } // gera 1 código de rastreio por palete físico (grupoPalete), não por // linha — várias linhas do mesmo palete misto (grupoPalete igual) // recebem o MESMO código; paletes diferentes (grupoPalete diferente) // recebem códigos diferentes, mesmo que entrem no mesmo lote const codigosJaUsados = {}; const dadosAtuais = aba.getDataRange().getValues(); for (let i = 1; i < dadosAtuais.length; i++) { const c = String(dadosAtuais[i][idx.codigoPalete] || '').trim(); if (c) codigosJaUsados[c] = true; } const codigoPorGrupo = {}; const codigosGerados = []; body.entradas.forEach(function (ent) { const grupo = (ent.grupoPalete === undefined || ent.grupoPalete === null) ? 0 : ent.grupoPalete; if (codigoPorGrupo[grupo] !== undefined) return; // já gerado pra esse palete físico const base = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd-HHmm'); let candidato = base; let sufixo = 1; while (codigosJaUsados[candidato]) { sufixo++; candidato = base + '-' + sufixo; } codigosJaUsados[candidato] = true; codigoPorGrupo[grupo] = candidato; codigosGerados.push(candidato); }); const linhasNovas = body.entradas.map(function (ent) { const grupo = (ent.grupoPalete === undefined || ent.grupoPalete === null) ? 0 : ent.grupoPalete; const linha = new Array(cabecalho.length).fill(''); linha[idx.familia] = ent.familia; linha[idx.produto] = ent.produto; linha[idx.rua] = ent.rua; linha[idx.lado] = ent.lado; linha[idx.andar] = ent.andar; linha[idx.posicao] = ent.posicao; linha[idx.palete] = 'P' + ent.palete; // planilha guarda "P1"/"P2", não só o número linha[idx.dataProducao] = ent.dataProducao; linha[idx.dataEntrada] = ent.dataEntrada; linha[idx.pacotes] = ent.pacotes; linha[idx.codigoPalete] = codigoPorGrupo[grupo]; return linha; }); // grava tudo de uma vez, logo abaixo da última linha usada aba.getRange(aba.getLastRow() + 1, 1, linhasNovas.length, cabecalho.length).setValues(linhasNovas); // se alguma família dessa entrada ainda não existe na aba FAMILIAS, // registra ela lá também — sem isso, a aba de referência fica // desatualizada toda vez que uma família nova nasce pela Nova Entrada const avisoFamilias = registrarFamiliasNovas(aba.getParent(), body.entradas); return respostaJson({ ok: true, linhas: linhasNovas.length, avisoFamilias: avisoFamilias, codigos: codigosGerados }); } /* Verifica quais famílias do lote ainda não existem na aba FAMILIAS (coluna FAMILIA) e adiciona uma linha pra cada uma — usando o código/produto/rua da própria entrada que criou a família. Se a aba FAMILIAS não existir ou faltar alguma coluna, não quebra a operação principal (os paletes já foram gravados) — só avisa no retorno. */ function registrarFamiliasNovas(ss, entradas) { const abaFamilias = ss.getSheetByName(NOME_ABA_FAMILIAS); if (!abaFamilias) return 'Aba "' + NOME_ABA_FAMILIAS + '" não encontrada — família não foi registrada lá.'; const dados = abaFamilias.getDataRange().getValues(); const cabecalho = dados[0]; const colFamilia = achaColuna(cabecalho, ['FAMÍLIA', 'FAMILIA']); const colCodigo = achaColuna(cabecalho, ['CÓDIGO', 'CODIGO']); const colProduto = cabecalho.indexOf('PRODUTO'); const colRua = cabecalho.indexOf('RUA'); if (colFamilia === -1 || colCodigo === -1 || colProduto === -1 || colRua === -1) { return 'Colunas da aba "' + NOME_ABA_FAMILIAS + '" não bateram com o esperado (FAMILIA/CÓDIGO/PRODUTO/RUA) — família não foi registrada lá.'; } const familiasExistentes = {}; for (let i = 1; i < dados.length; i++) { const nome = String(dados[i][colFamilia] || '').trim(); if (nome) familiasExistentes[nome] = true; } // agrupa as entradas recebidas por família (uma família pode ter usado // mais de uma rua no mesmo lote, se a quantidade não coube numa só) const porFamilia = {}; entradas.forEach(function (ent) { const nome = String(ent.familia || '').trim(); if (!nome || familiasExistentes[nome]) return; // já existe, não repete if (!porFamilia[nome]) porFamilia[nome] = { codigo: ent.codigo || '', produto: ent.produto || '', ruas: new Set() }; porFamilia[nome].ruas.add(String(ent.rua)); }); const linhasNovas = Object.keys(porFamilia).map(function (nome) { const info = porFamilia[nome]; const linha = new Array(cabecalho.length).fill(''); linha[colFamilia] = nome; linha[colCodigo] = info.codigo; linha[colProduto] = info.produto; linha[colRua] = [...info.ruas].sort().join(';'); return linha; }); if (linhasNovas.length === 0) return null; abaFamilias.getRange(abaFamilias.getLastRow() + 1, 1, linhasNovas.length, cabecalho.length).setValues(linhasNovas); return null; } // Índices de todas as colunas usadas pelas duas ações, achados pelo nome // do cabeçalho — assim continua funcionando mesmo se a ordem das colunas // na planilha mudar. function idxColunas(cabecalho) { return { familia: achaColuna(cabecalho, ['FAMÍLIA', 'FAMILIA']), produto: cabecalho.indexOf('PRODUTO'), rua: cabecalho.indexOf('RUA'), lado: cabecalho.indexOf('LADO'), andar: cabecalho.indexOf('ANDAR'), posicao: achaColuna(cabecalho, ['POSIÇÃO', 'POSICAO']), palete: cabecalho.indexOf('PALETE'), dataProducao: achaColuna(cabecalho, ['DATA DE PRODUÇÃO', 'DATA DE PRODUCAO', 'DATA PRODUCAO']), dataEntrada: cabecalho.indexOf('DATA ENTRADA'), dataSaida: achaColuna(cabecalho, ['DATA SAÍDA', 'DATA SAIDA']), pacotes: cabecalho.indexOf('PACOTES'), codigoPalete: achaColuna(cabecalho, ['CÓDIGO DE RASTREIO', 'CODIGO DE RASTREIO', 'CÓDIGO PALETE', 'CODIGO PALETE']), }; } // Acha a linha exata do registro. Prioriza o número da linha real da // planilha (p.linha), mandado pelo painel — é confiável mesmo quando duas // linhas diferentes têm o mesmo endereço por erro de apontamento, porque // não depende de casar rua/lado/andar/posição/palete pra achar a linha // certa. Se p.linha não vier (painel antigo em cache em algum navegador), // cai pro jeito antigo de achar por endereço, como rede de segurança. function acharLinhaDoRegistro(dados, idx, p) { if (p.linha) { const i = p.linha - 1; // p.linha é o número da linha na planilha (1 = cabeçalho); dados[] é 0-based com dados[0] = cabeçalho if (i >= 1 && i < dados.length) { const jaTemSaida = String(dados[i][idx.dataSaida] || '').trim() !== ''; if (!jaTemSaida) return i; } return -1; // linha não existe mais ou já foi processada — não tenta adivinhar por endereço } return acharLinhaLivre(dados, idx, p); } // Acha a primeira linha cujo endereço bate com o palete pedido e que // AINDA NÃO tem DATA SAÍDA preenchida (senão já estaria livre/retirada) // — usado só como rede de segurança quando o painel não manda p.linha. function acharLinhaLivre(dados, idx, p) { for (let i = 1; i < dados.length; i++) { const row = dados[i]; const bate = String(row[idx.rua]).trim() === String(p.rua).trim() && String(row[idx.lado]).trim().toUpperCase() === String(p.lado).trim().toUpperCase() && String(row[idx.andar]).trim() === String(p.andar).trim() && String(row[idx.posicao]).trim() === String(p.posicao).trim() && String(row[idx.palete]).replace(/[^0-9]/g, '') === String(p.palete).replace(/[^0-9]/g, ''); const jaTemSaida = String(row[idx.dataSaida] || '').trim() !== ''; if (bate && !jaTemSaida) return i; } return -1; } function achaColuna(cabecalho, nomesPossiveis) { for (let i = 0; i < nomesPossiveis.length; i++) { const pos = cabecalho.indexOf(nomesPossiveis[i]); if (pos !== -1) return pos; } return -1; } function enderecoTexto(p) { const ladoTxt = p.lado === 'U' ? '' : ' Lado ' + p.lado; return 'Rua ' + p.rua + ladoTxt + ' And.' + p.andar + ' Pos.' + p.posicao + '-P' + p.palete; } function respostaJson(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
